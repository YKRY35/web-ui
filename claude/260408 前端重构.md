  def load_config(self, config_path: str):
      """加载配置并直接更新Gradio UI组件"""
      # 1. 读取JSON配置文件
      with open(config_path, "r") as fr:
          ui_settings = json.load(fr)

      # 2. 遍历配置文件中的键值对
      for comp_id, comp_val in ui_settings.items():
          # 3. 通过组件ID获取对应的Gradio组件对象
          if comp_id in self.id_to_component:
              comp = self.id_to_component[comp_id]
              value = comp.__class__(value=comp_val)

              # 特殊处理planner_llm_provider字段
              if comp_id == "agent_settings.planner_llm_provider":
                  yield {comp: value}
                  time.sleep(0.1)

      # 4. 更新状态信息
      config_status = self.id_to_component["load_save_config.config_status"]
      yield {config_status: f"Successfully loaded config: {config_path}"}

  def save_config(self, components: Dict["Component", str]) -> None:
      """保存当前UI组件状态到配置文件"""
      cur_settings = {}

      # 1. 遍历所有组件，排除按钮和文件选择器
      for comp in components:
          if not isinstance(comp, gr.Button) and not isinstance(comp, gr.File):
              comp_id = self.get_id_by_component(comp)
              cur_settings[comp_id] = components[comp]

      # 2. 保存为JSON文件，使用时间戳命名
      config_name = datetime.now().strftime("%Y%m%d-%H%M%S")
      config_path = os.path.join(self.settings_save_dir, f"{config_name}.json")

      with open(config_path, "w") as fw:
