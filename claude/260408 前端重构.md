  3. API设计

  后端使用FastAPI/Flask提供RESTful API，前端通过axios请求：

  3.1 配置相关API

  // 获取/更新代理设置
  GET    /api/agent-settings
  POST   /api/agent-settings

  // 获取/更新浏览器设置
  GET    /api/browser-settings
  POST   /api/browser-settings

  // 加载配置
  POST   /api/load-config
  { file: File }

  // 保存配置
  GET    /api/save-config

  3.2 代理运行API

  // 运行浏览器使用代理
  POST   /api/agent/run
  {
    task: string,
    settings: AgentSettings & BrowserSettings
  }
  // 返回：{ task_id: string }

  // 停止代理
  POST   /api/agent/stop
  { task_id: string }

  // 暂停/恢复代理
  POST   /api/agent/pause
  { task_id: string }

  POST   /api/agent/resume
  { task_id: string }

  // 获取代理状态
  GET    /api/agent/status?task_id={task_id}
  // 返回：{ status: 'running'|'paused'|'stopped', chat_history: [], browser_view: string }

  // 发送用户响应（当代理需要帮助时）
  POST   /api/agent/respond
  {
    task_id: string,
    message: string
  }

  3.3 深度研究API

  // 运行深度研究
  POST   /api/deep-research/run
  {
    topic: string,
    resume_task_id?: string,
    parallel_num: number,
    save_dir: string,
    settings: AgentSettings & BrowserSettings
  }
  // 返回：{ task_id: string }

  // 停止深度研究
  POST   /api/deep-research/stop
  { task_id: string }

  // 获取研究报告
  GET    /api/deep-research/report?task_id={task_id}
  // 返回：{ content: string }
